METRICS_DIR="./metrics"
echo "date,success_rate,p99_latency_ms,token_refreshes,panic_count,memory_mb"
for i in $(seq 1 7); do
  d=$(date -v-${i}d +%Y-%m-%d 2>/dev/null || date -d "-${i} days" +%Y-%m-%d)
  f="$METRICS_DIR/metrics-$d.json"
  if [ -f "$f" ]; then
    echo -n "$d,"
    cat "$f" | grep -oE '"success_rate":\s*[0-9.]+' | head -1 | cut -d: -f2 | tr -d ' '
    echo -n ","
    cat "$f" | grep -oE '"p99_latency_ms":\s*[0-9]+' | head -1 | cut -d: -f2 | tr -d ' '
    echo ""
  fi
done
